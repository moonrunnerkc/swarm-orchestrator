/**
 * Behaviors to check:
 * 1. --reviewers option takes a comma-separated list of short names: tech-debt, docs and security.
 * 2. parseArgs returns the choice as `reviewers`, the full agent names in the order tech-debt, docs, security whatever order they were typed, each once even when repeated.
 * 3. Without the flag, leave `reviewers` unset.
 * 4. A missing or empty value is an error naming the valid short names (tech-debt, docs, security).
 * 5. A name that isn't one of the three is an error naming it and the valid short names (tech-debt, docs, security).
 * 6. orchestrate takes the same list as an optional `reviewers` param (all three when omitted).
 * 7. main passes the parsed list through to orchestrate.
 * 8. Document the flag in the --help text.
 */
import { execFileSync } from "node:child_process";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import {
  afterEach,
  beforeEach,
  describe,
  expect,
  it,
  vi,
  type MockInstance,
} from "vitest";

vi.mock("../src/claude-code.js", () => ({
  findClaudeCodeBinary: vi.fn(() => null),
}));

vi.mock("../src/orchestrate.js", () => ({
  orchestrate: vi.fn(),
}));

import { findClaudeCodeBinary } from "../src/claude-code.js";
import { main, parseArgs } from "../src/cli.js";
import { orchestrate } from "../src/orchestrate.js";

const findBinaryMock = vi.mocked(findClaudeCodeBinary);
const orchestrateMock = vi.mocked(orchestrate);

function makeCleanRepo(): string {
  const dir = mkdtempSync(join(tmpdir(), "pubprep-cli-"));
  execFileSync("git", ["init", "-q", "-b", "main"], { cwd: dir });
  execFileSync("git", ["config", "user.email", "test@example.com"], { cwd: dir });
  execFileSync("git", ["config", "user.name", "Test"], { cwd: dir });
  writeFileSync(join(dir, ".gitignore"), ".pubprep/\n.env\n");
  writeFileSync(join(dir, "README.md"), "# fixture\n");
  execFileSync("git", ["add", "."], { cwd: dir });
  execFileSync("git", ["commit", "-q", "-m", "init"], { cwd: dir });
  return dir;
}

function successfulOrchestrate() {
  return Promise.resolve({
    runDir: "/tmp/run",
    manifestPath: "/tmp/run/manifest.json",
    exitReason: "success",
    manifest: {
      run_id: "test-run",
      started_at: "2026-05-22T00:00:00Z",
      finished_at: "2026-05-22T00:01:00Z",
      target_repo: "/tmp/run",
      target_head_sha: "deadbeef",
      agents: [],
      convergence_branch: null,
      exit_reason: "success",
      publish_gate: null,
      pull_request: null,
      warnings: [],
    },
  });
}

describe("parseArgs()", () => {
  it("returns full agent names in the order tech-debt, docs, security regardless of input order", () => {
    const args = ["--reviewers", "security,docs,tech-debt"];
    const result = parseArgs(args);
    expect(result.reviewers).toEqual([
      "tech-debt-reviewer",
      "readme-docs-reviewer",
      "security-reviewer",
    ]);
  });

  it("returns each reviewer only once even when repeated", () => {
    const args = ["--reviewers", "docs,tech-debt,docs,security,tech-debt"];
    const result = parseArgs(args);
    expect(result.reviewers).toEqual([
      "tech-debt-reviewer",
      "readme-docs-reviewer",
      "security-reviewer",
    ]);
  });

  it("leaves reviewers unset when the flag is absent", () => {
    const result = parseArgs([]);
    expect(result.reviewers).toBeUndefined();
  });

  it("throws an error naming valid short names when value is missing", () => {
    expect(() => parseArgs(["--reviewers"])).toThrow(
      expect.stringMatching(/tech-debt.*docs.*security/)
    );
  });

  it("throws an error naming valid short names when value is empty", () => {
    expect(() => parseArgs(["--reviewers", ""])).toThrow(
      expect.stringMatching(/tech-debt.*docs.*security/)
    );
  });

  it("throws an error naming the invalid name and valid short names when an invalid name is provided", () => {
    const invalid = "unknown-reviewer";
    expect(() => parseArgs(["--reviewers", invalid])).toThrow(
      expect.stringMatching(new RegExp(`${invalid}.*tech-debt.*docs.*security`))
    );
  });
});

describe("main() and orchestrate() plumbing", () => {
  let stdoutSpy: MockInstance<(chunk: string) => boolean>;
  let stderrSpy: MockInstance<(chunk: string) => boolean>;
  let originalApiKey: string | undefined;
  let repo: string | null;
  let fakeHome: string;

  beforeEach(() => {
    stdoutSpy = vi
      .spyOn(process.stdout, "write")
      .mockImplementation(() => true) as MockInstance<(chunk: string) => boolean>;
    stderrSpy = vi
      .spyOn(process.stderr, "write")
      .mockImplementation(() => true) as MockInstance<(chunk: string) => boolean>;
    originalApiKey = process.env.ANTHROPIC_API_KEY;
    delete process.env.ANTHROPIC_API_KEY;
    findBinaryMock.mockReset();
    findBinaryMock.mockReturnValue(null);
    orchestrateMock.mockReset();
    orchestrateMock.mockImplementation(successfulOrchestrate);
    repo = null;
    fakeHome = mkdtempSync(join(tmpdir(), "pubprep-cli-home-"));
  });

  afterEach(() => {
    stdoutSpy.mockRestore();
    stderrSpy.mockRestore();
    if (originalApiKey === undefined) {
      delete process.env.ANTHROPIC_API_KEY;
    } else {
      process.env.ANTHROPIC_API_KEY = originalApiKey;
    }
    if (repo !== null) {
      rmSync(repo, { recursive: true, force: true });
    }
    rmSync(fakeHome, { recursive: true, force: true });
  });

  function stdoutText(): string {
    return stdoutSpy.mock.calls.map((c) => String(c[0])).join("");
  }

  it("passes the parsed reviewers list to orchestrate", async () => {
    repo = makeCleanRepo();
    findBinaryMock.mockReturnValue("/opt/homebrew/bin/claude");
    await main(["--reviewers", "docs,security"], { cwd: repo, home: fakeHome });
    expect(orchestrateMock).toHaveBeenCalledWith(
      expect.objectContaining({
        reviewers: ["readme-docs-reviewer", "security-reviewer"],
      })
    );
  });

  it("does not pass reviewers to orchestrate when flag is absent", async () => {
    repo = makeCleanRepo();
    findBinaryMock.mockReturnValue("/opt/homebrew/bin/claude");
    await main([], { cwd: repo, home: fakeHome });
    const call = orchestrateMock.mock.calls[0]?.[0];
    expect(call?.reviewers).toBeUndefined();
  });

  it("documents the --reviewers flag in the --help text", async () => {
    await main(["--help"]);
    const out = stdoutText();
    expect(out).toMatch(/--reviewers/);
    expect(out).toMatch(/tech-debt/);
    expect(out).toMatch(/docs/);
    expect(out).toMatch(/security/);
  });
});
