    const outDir = tsconfig.compilerOptions.outDir;

      `${chalk.red('ERROR:')} The ${chalk.gray('clean')} command` +
        ` requires ${chalk.gray('compilerOptions.outDir')} to be defined in ` +
        'tsconfig.json.',
    );
    return false;
  }
}
